# Hálózati Topológia

Két Docker network:

1. `app-net` (bridge): nginx, api, websocket, localstack
2. `data-net` (bridge, internal: true): postgres, redis, api, websocket, worker, localstack

Az api, websocket és localstack MINDKÉT hálózaton rajta van.
A worker CSAK data-net-en van (nincs publikus portja).
A postgres és redis CSAK data-net-en van, de a "ports:" mapping
miatt a host gépről elérhetők fejlesztéshez.

## 📄 backend/Dockerfile (Production multi-stage)

# STAGE 1: Dependencies
FROM golang:1.26.1-alpine3.21 AS deps
WORKDIR /build
RUN apk add --no-cache ca-certificates tzdata
COPY go.mod go.sum ./
RUN go mod download && go mod verify

# STAGE 2: Builder
FROM deps AS builder
ARG SERVICE=api
COPY . .
RUN CGO_ENABLED=0 GOOS=linux GOARCH=amd64 \
    go build -ldflags="-s -w" -trimpath \
    -o /build/bin/service ./cmd/${SERVICE}/main.go

# STAGE 3: Runtime
FROM alpine:3.21 AS runtime
ARG SERVICE
LABEL service="${SERVICE}"
COPY --from=builder /etc/ssl/certs/ca-certificates.crt /etc/ssl/certs/
COPY --from=builder /usr/share/zoneinfo /usr/share/zoneinfo
RUN addgroup -g 10001 -S appgroup && \
    adduser -u 10001 -S appuser -G appgroup -h /app -s /sbin/nologin
WORKDIR /app
COPY --from=builder /build/bin/service ./service
RUN chown -R appuser:appgroup /app && chmod +x /app/service
USER appuser:appgroup
EXPOSE 8080
HEALTHCHECK --interval=15s --timeout=5s --start-period=10s --retries=3 \
    CMD wget -qO- http://localhost:8080/healthz || exit 1
ENTRYPOINT ["./service"]

		RedisURL:         getEnv("REDIS_URL", "redis://localhost:6379/0"),

		AWSEndpointURL:   getEnv("AWS_ENDPOINT_URL", "http://localhost:4566"),
		AWSRegion:        getEnv("AWS_REGION", "us-east-1"),
		AWSAccessKey:     getEnv("AWS_ACCESS_KEY_ID", "test"),

func getEnv(key, fallback string) string {
	if val, ok := os.LookupEnv(key); ok {
		return val
	}
	return fallback

## 📄 backend/internal/aws/client.go

Ez egy KÖZPONTI AWS kliens inicializáló modul.
Ha az AWS_ENDPOINT_URL env var be van állítva (LocalStack),
az összes AWS SDK hívás oda megy. Ha nincs (production),
az AWS SDK automatikusan a valódi AWS-t használja.

A fájl tartalma:

package aws

import (
	"context"
	"log"

	"github.com/aws/aws-sdk-go-v2/aws"
	"github.com/aws/aws-sdk-go-v2/config"
	"github.com/aws/aws-sdk-go-v2/credentials"
	awsconfig "github.com/user/fitchallenge/internal/config"
)

func NewAWSConfig(ctx context.Context, cfg *awsconfig.Config) (aws.Config, error) {
	var opts []func(*config.LoadOptions) error

	opts = append(opts, config.WithRegion(cfg.AWSRegion))

	// Ha van explicit credentials (LocalStack-hez "test"/"test")
	if cfg.AWSAccessKey != "" {
		opts = append(opts, config.WithCredentialsProvider(
			credentials.NewStaticCredentialsProvider(cfg.AWSAccessKey, cfg.AWSSecretKey, ""),
		))
	}

	awsCfg, err := config.LoadDefaultConfig(ctx, opts...)
	if err != nil {
		return aws.Config{}, err
	}

	// Ha LocalStack endpoint van megadva, MINDEN service oda megy
	if cfg.AWSEndpointURL != "" {
		awsCfg.BaseEndpoint = aws.String(cfg.AWSEndpointURL)
		log.Printf("AWS SDK configured with LocalStack endpoint: %s", cfg.AWSEndpointURL)
	}

	return awsCfg, nil
}

## 📄 backend/internal/pkg/response/json.go

package response

import (
	"time"

	"github.com/gin-gonic/gin"
)

type SuccessResponse struct {
	Success bool        `json:"success"`
	Data    interface{} `json:"data"`
	Meta    Meta        `json:"meta"`
}

type ErrorResponse struct {
	Success bool      `json:"success"`
	Error   ErrorBody `json:"error"`
	Meta    Meta      `json:"meta"`
}

type ErrorBody struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

type Meta struct {
	Timestamp string `json:"timestamp"`
}

func Success(c *gin.Context, status int, data interface{}) {
	c.JSON(status, SuccessResponse{
		Success: true,
		Data:    data,
		Meta:    Meta{Timestamp: time.Now().UTC().Format(time.RFC3339)},
	})
}

func Error(c *gin.Context, status int, code string, message string) {
	c.JSON(status, ErrorResponse{
		Success: false,
		Error:   ErrorBody{Code: code, Message: message},
		Meta:    Meta{Timestamp: time.Now().UTC().Format(time.RFC3339)},
	})
}


## 📄 backend/internal/handler/http/health_handler.go

package http

import (
	"context"
	"net/http"
	"time"

	"github.com/gin-gonic/gin"
	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/redis/go-redis/v9"

	"github.com/user/fitchallenge/internal/pkg/response"
)

type HealthHandler struct {
	db  *pgxpool.Pool
	rdb *redis.Client
	svc string
}

func NewHealthHandler(db *pgxpool.Pool, rdb *redis.Client, serviceName string) *HealthHandler {
	return &HealthHandler{db: db, rdb: rdb, svc: serviceName}
}

func (h *HealthHandler) HealthCheck(c *gin.Context) {
	response.Success(c, http.StatusOK, gin.H{
		"status":    "ok",
		"service":   h.svc,
		"timestamp": time.Now().UTC().Format(time.RFC3339),
	})
}

func (h *HealthHandler) ReadinessCheck(c *gin.Context) {
	ctx, cancel := context.WithTimeout(c.Request.Context(), 3*time.Second)
	defer cancel()

	pgStatus := "connected"
	redisStatus := "connected"
	status := http.StatusOK

	if h.db != nil {
		if err := h.db.Ping(ctx); err != nil {
			pgStatus = "disconnected: " + err.Error()
			status = http.StatusServiceUnavailable
		}
	} else {
		pgStatus = "not configured"
	}

	if h.rdb != nil {
		if err := h.rdb.Ping(ctx).Err(); err != nil {
			redisStatus = "disconnected: " + err.Error()
			status = http.StatusServiceUnavailable
		}
	} else {
		redisStatus = "not configured"
	}

	if status == http.StatusOK {
		response.Success(c, status, gin.H{
			"status":   "ok",
			"postgres": pgStatus,
			"redis":    redisStatus,
		})
	} else {
		response.Error(c, status, "NOT_READY", "One or more dependencies unavailable")
	}
}

A script tartalma:

#!/bin/bash
echo "=== Initializing LocalStack AWS resources ==="

# S3: Bucket létrehozás
echo "Creating S3 bucket..."
awslocal s3 mb s3://fitchallenge-assets
awslocal s3api put-bucket-cors --bucket fitchallenge-assets --cors-configuration '{
  "CORSRules": [
    {
      "AllowedOrigins": ["*"],
      "AllowedMethods": ["GET", "PUT", "POST"],
      "AllowedHeaders": ["*"],
      "MaxAgeSeconds": 3600
    }
  ]
}'
echo "S3 bucket ready: fitchallenge-assets"

# SQS: Queue létrehozás
echo "Creating SQS queues..."
awslocal sqs create-queue --queue-name fitchallenge-jobs
awslocal sqs create-queue --queue-name fitchallenge-jobs-dlq
echo "SQS queues ready"

# SNS: Topic létrehozás
echo "Creating SNS topics..."
awslocal sns create-topic --name fitchallenge-notifications
echo "SNS topics ready"

# SES: Email identity verifikáció
echo "Verifying SES email identities..."
awslocal ses verify-email-identity --email-address noreply@fitchallenge.local
echo "SES identities verified"

# Secrets Manager: Alap titkosított értékek
echo "Creating secrets..."
awslocal secretsmanager create-secret \
    --name fitchallenge/jwt-secret \
    --secret-string '{"key":"local-dev-secret-change-this-in-production-min32chars"}'
awslocal secretsmanager create-secret \
    --name fitchallenge/db-credentials \
    --secret-string '{"username":"fc_user","password":"fc_local_password_123","host":"postgres","port":"5432","dbname":"fitchallenge"}'
echo "Secrets created"

# Ellenőrzés
echo ""
echo "=== LocalStack AWS Resources Summary ==="
echo "S3 Buckets:"
awslocal s3 ls
echo ""
echo "SQS Queues:"
awslocal sqs list-queues
echo ""
echo "SNS Topics:"
awslocal sns list-topics
echo ""
echo "Secrets:"
awslocal secretsmanager list-secrets --query 'SecretList[].Name' --output text
echo ""
echo "=== LocalStack initialization complete ==="

# Local Development Setup

## Prerequisites

- Docker Desktop (v4.0+)
- Make (usually pre-installed on macOS/Linux)
- Git
- LocalStack Student account (free via GitHub Student Developer Pack)

## LocalStack Setup (FONTOS — ELSŐ LÉPÉS!)

1. Regisztrálj a LocalStack Student plan-re:
   https://www.localstack.cloud/localstack-for-students

2. Kérj Auth Token-t:
   https://app.localstack.cloud → Settings → Auth Token

3. Másold be a token-t a .env fájlba:
   LOCALSTACK_AUTH_TOKEN=your-token-here

## First Time Setup

1. Clone the repository:
   git clone <repo-url>
   cd community-fitness-challenge

2. Copy and edit .env:
   cp .env.example .env
   # EDIT .env → add your LOCALSTACK_AUTH_TOKEN!

3. Run setup:
   make setup

4. Verify:
   make status
   make api-health
   make aws-status

## Daily Usage

Start: make dev
Stop:  make stop
Logs:  make logs

## Available URLs

| Service         | URL                        |
|-----------------|----------------------------|
| API (direct)    | http://localhost:8080       |
| API (via Nginx) | http://localhost:80         |
| WebSocket       | ws://localhost:8081         |
| PostgreSQL      | localhost:5432              |
| Redis           | localhost:6379              |
| LocalStack      | http://localhost:4566       |
| LocalStack Web  | https://app.localstack.cloud|

## AWS Services (via LocalStack)

All AWS services run inside LocalStack on port 4566.
Use endpoint: http://localhost:4566

Available services:
- S3: Avatar images, challenge covers
- SQS: Background job queue
- SNS: Push notifications
- SES: Email sending
- Secrets Manager: Credential storage
- CloudWatch: Logging
- Lambda: Serverless functions
- IAM: Access control
- KMS: Encryption keys

Test AWS services:
  make aws-s3-ls            # List S3 buckets
  make aws-s3-upload-test   # Upload test file
  make aws-sqs-send-test    # Send test SQS message
  make aws-ses-test         # Send test email
  make aws-secrets-ls       # List secrets

## Hot Reload

All three Go services use Air for hot-reload:
  - API:       .air.api.toml    (builds cmd/api/main.go)
  - WebSocket: .air.ws.toml     (builds cmd/ws/main.go)
  - Worker:    .air.worker.toml (builds cmd/worker/main.go)

## Troubleshooting

LocalStack not starting:
  Check that LOCALSTACK_AUTH_TOKEN is set in .env.
  Run: make logs-localstack

Port conflict:
  Change NGINX_PORT in .env.

Build fails on first run:
  First "go mod download" may take 1-2 minutes.
  If it fails: make nuke && make setup

All else fails:
  make nuke
  make setup
